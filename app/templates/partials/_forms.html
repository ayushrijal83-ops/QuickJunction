{# Form-field macros. Bootstrap validation feedback is driven by the *server's* WTForms errors (`is-invalid` + `invalid-feedback`), not by client-side validation, so the feedback a user sees is always the same judgement the server made. Forms carry `novalidate` for the same reason. #} {% macro field(f, autofocus=false, help=none, type=none) %}
{{ f.label(class="form-label") }} {{ f(class="form-control" + (" is-invalid" if f.errors else ""), autofocus=autofocus, **kwargs) }} {% if help %}
{{ help }}
{% endif %} {% for error in f.errors %}
{{ error }}
{% endfor %}
{% endmacro %} {% macro select(f, help=none) %}
{{ f.label(class="form-label") }} {{ f(class="form-select" + (" is-invalid" if f.errors else ""), **kwargs) }} {% if help %}
{{ help }}
{% endif %} {% for error in f.errors %}
{{ error }}
{% endfor %}
{% endmacro %} {% macro checkbox(f, help=none) %}
{{ f(class="form-check-input" + (" is-invalid" if f.errors else "")) }} {{ f.label(class="form-check-label") }} {% if help %}
{{ help }}
{% endif %} {% for error in f.errors %}
{{ error }}
{% endfor %}
